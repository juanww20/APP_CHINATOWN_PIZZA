const Tab = createBottomTabNavigator();
const Stack = createNativeStackNavigator();

function StartShoppingScreen({ onStart }) {

  const [text, setText] = React.useState("");

  return (
    <View style={styles.startContainer}>
      <StatusBar barStyle="light-content" backgroundColor="#1A1A1A" />
      <Text style={styles.title}>¡Callejera, sabrosa y auténtica pizza! 🍕</Text>
      <View style={styles.caja_imagen}>
        <Image source={pizza_icono} style={styles.imagen} />
      </View>
      <TextInput
        style={styles.input}
        placeholder='Ingresa tu Nombre'
        value={text}
        onChangeText={setText} // Usamos onChangeText en lugar de onChange
        placeholderTextColor={'gray'}
      />
      <TouchableOpacity onPress={onStart} style={styles.button}>
        <Text style={styles.buttonText}>Pedir PIZZA OWO</Text>
      </TouchableOpacity>
      <Text style={styles.infotext}>Estás en la mesa #1</Text>
    </View>
  );
}

export default function App() {

  const [isStarted, setIsStarted] = useState(false);

  if (!isStarted) {
    return (
      <StartShoppingScreen onStart={() => setIsStarted(true)} />
    );
  }

  return (
    <NavigationContainer>
      <Tab.Navigator screenOptions={({ route }) => ({
        tabBarActiveTintColor: '#2196F3',
        tabBarInactiveTintColor: 'gray',
        tabBarIcon: ({ focused, color, size }) => {
          let iconName;

          if (route.name === 'Menu') {
            iconName = focused ? 'pizza' : 'pizza-outline';
          } else if (route.name === 'Carrito') {
            iconName = focused ? 'cart' : 'cart-outline';
          }

          return <Icon name={iconName} size={size} color={color} />;
        },
      })}>
        <Tab.Screen name='Menu' component={Menu_general} options={{ headerShown: false }} />
        <Tab.Screen name='Carrito' component={Carrito} options={{ headerShown: false }} />
      </Tab.Navigator>
    </NavigationContainer>
  );
}
